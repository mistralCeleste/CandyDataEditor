        Html

    [Parameter] public EventCallback<string> MarkdownValueChanged { get; set; }

    private bool _isInitialized = false;
    private List<string>? _lastSyncedDictionary = null;

    // View Switching logic
    private async Task SwitchViewMode(ViewMode newMode)
    {
        if (currentViewMode == newMode) return;

        currentViewMode = newMode;

        // Hide right-click spellcheck menu when leaving Formatted mode
        isContextMenuVisible = false;

        // If returning back to Formatted View, update TipTap instance content with the latest edits
        if (newMode == ViewMode.Formatted && _isInitialized)
        {
            await JSRuntime.InvokeVoidAsync("setTipTapContentFromMarkdown", EditorId, MarkdownValue);
        }
    }

    // Direct Markdown Textarea Edits
    private async Task OnMarkdownRawInput(ChangeEventArgs e)
    {
        string updatedMarkdown = e.Value?.ToString() ?? string.Empty;
        MarkdownValue = updatedMarkdown;
        await MarkdownValueChanged.InvokeAsync(updatedMarkdown);

        // Keep TipTap document synchronized in background
        if (_isInitialized)
        {
            await JSRuntime.InvokeVoidAsync("setTipTapContentFromMarkdown", EditorId, updatedMarkdown);
        }
    }

    // Direct HTML Textarea Edits
    private async Task OnHtmlRawInput(ChangeEventArgs e)
    {
        string updatedHtml = e.Value?.ToString() ?? string.Empty;
        Value = updatedHtml;
        await ValueChanged.InvokeAsync(updatedHtml);

        if (_isInitialized)
        {
            await JSRuntime.InvokeVoidAsync("setTipTapContentFromHtml", EditorId, updatedHtml);
        }
    }

    [JSInvokable]
    public async Task OnContentChanged(string html, string markdown)
    {
        Value = html;
        await ValueChanged.InvokeAsync(html);

        MarkdownValue = markdown;
        await MarkdownValueChanged.InvokeAsync(markdown);
    }

    // ... Keep existing OnAfterRenderAsync, OnParametersSetAsync, Context Menu methods intact ...
}
    [JSInvokable]
    public async Task OnContentChanged(string newHtml, string newMarkdown)
    {
        if (Value != newHtml)
        {
            Value = newHtml;
            await ValueChanged.InvokeAsync(Value);
        }

        if (MarkdownValue != newMarkdown)
        {
            MarkdownValue = newMarkdown;
            await MarkdownValueChanged.InvokeAsync(MarkdownValue);
        }
    }
