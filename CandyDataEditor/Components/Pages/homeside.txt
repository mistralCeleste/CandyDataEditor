<!-- Home.razor -->
@page "/"
@using CandyDataEditor.Components
@using CandyDataEditor.Services
@inject GameDictionaryService DictionaryService

<PageTitle>Game Data Editor</PageTitle>

<div class="container my-4">
    <h2>Game Data Editor Test</h2>
    <p class="text-muted">Editing Item Description with TipTap v3 & Markdown</p>

    <!-- 1. Dictionary Quick-Add Bar -->
    <div class="card mb-3 bg-light">
        <div class="card-body d-flex gap-2 align-items-center">
            <span class="fw-bold">Add Custom Term:</span>
            <input type="text" class="form-control form-control-sm w-25" 
                   placeholder="e.g. Lyratha, Mithril..." 
                   @bind="newCustomWord" 
                   @keyup="HandleKeyUp" />
            
            <button class="btn btn-sm btn-success" @onclick="AddNewWordToDictionary">
                ➕ Add to Dictionary
            </button>
            
            <span class="text-muted ms-auto">
                <small>Active File: <code>@activeDictionaryFileName</code> (@activeDictionary.Count terms)</small>
            </span>
        </div>
    </div>

    <!-- 2. TipTap Editor -->
    <div class="card mb-4">
        <div class="card-header font-weight-bold d-flex justify-content-between align-items-center">
            <span>Item Description</span>
        </div>
        <div class="card-body">
            <TipTapEditor @bind-Value="gameItemHtml" 
                          @bind-MarkdownValue="gameItemMarkdown" 
                          GameDictionaryWords="activeDictionary" />
        </div>
    </div>

    <!-- 3. Controls & Actions -->
    <div class="d-flex gap-2 mb-4">
        <button class="btn btn-secondary" @onclick="ResetContent">Reset Text</button>
        <button class="btn btn-outline-primary" @onclick="InsertSampleLore">Insert Game Lore</button>
    </div>

    <!-- 4. Live Debug Views -->
    <div class="row">
        <!-- Raw Markdown Output -->
        <div class="col-md-6 mb-3">
            <div class="card border-success">
                <div class="card-header bg-success text-white">
                    <strong>Raw Markdown Output (Database State)</strong>
                </div>
                <div class="card-body bg-light">
                    <pre class="mb-0"><code>@gameItemMarkdown</code></pre>
                </div>
            </div>
        </div>

        <!-- Rendered HTML Output -->
        <div class="col-md-6 mb-3">
            <div class="card border-info">
                <div class="card-header bg-info text-white">
                    <strong>HTML Output</strong>
                </div>
                <div class="card-body bg-light">
                    <pre class="mb-0"><code>@gameItemHtml</code></pre>
                </div>
            </div>
        </div>
    </div>
</div>

@code {
    private string gameItemHtml = "<p>The <strong>Lyratha warblade</strong> grants +100 attack.</p>";
    private string gameItemMarkdown = string.Empty;

    // Use filename only — Service builds the safe writeable path!
    private string activeDictionaryFileName = "fantasy.txt";
    private List<string> activeDictionary = new();
    private string newCustomWord = string.Empty;

    protected override async Task OnInitializedAsync()
    {
        // Load words from dictionary file on startup
        activeDictionary = await DictionaryService.LoadDictionaryAsync(activeDictionaryFileName);
    }

    private async Task AddNewWordToDictionary()
    {
        if (string.IsNullOrWhiteSpace(newCustomWord)) return;

        var cleanWord = newCustomWord.Trim();

        // 1. Save to local AppData folder
        await DictionaryService.AddWordToDictionaryFileAsync(activeDictionaryFileName, cleanWord);

        // 2. Refresh active list (new List instance forces Blazor Parameter update)
        if (!activeDictionary.Contains(cleanWord, StringComparer.OrdinalIgnoreCase))
        {
            activeDictionary = new List<string>(activeDictionary) { cleanWord };
        }

        newCustomWord = string.Empty; // Clear text field
    }

    private async Task HandleKeyUp(KeyboardEventArgs e)
    {
        if (e.Key == "Enter")
        {
            await AddNewWordToDictionary();
        }
    }

    private void ResetContent()
    {
        gameItemHtml = "<p>Default game item description...</p>";
    }

    private void InsertSampleLore()
    {
        gameItemHtml = "<h3>Excalibur</h3><p>Forged in ancient times by <b>Lyratha</b>, this blade inflicts <b>100 holy damage</b>.</p>";
    }
}
